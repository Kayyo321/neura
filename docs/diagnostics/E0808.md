# E0808: generic specialization limit exceeded

Generic specialization makes a copy of each generic declaration per set of type arguments; a chain of instantiations whose type arguments keep growing never ends, and is stopped at a nesting limit.

<!-- example: error: E0808 -->
```neura
let Box = class<T> (
    export let value: T;
    export let init = func(value: T) {
        instance.value = value;
    };
);
let wrap = func<T>(value: T, depth: int) -> int {
    if (depth == 0) {
        return 0;
    }
    return wrap<Box<T>>(new Box<T>(value), depth - 1);
};
let program = import:Std.Program;

export let Main = module : program.Entry (
    export let.override main = func(arguments: string[]) -> program.ExitCode {
        wrap<int>(1, 3);
        return program.ExitCode.Ok;
    };
);
```

Recurse with the same type arguments, and keep the growing structure in a value:

<!-- example: run -->
```neura
let wrap = func<T>(value: T, depth: int) -> int {
    if (depth == 0) {
        return 0;
    }
    return wrap<T>(value, depth - 1) + 1;
};
let console = import:Std.Io.Console;
let program = import:Std.Program;

export let Main = module : program.Entry (
    export let.override main = func(arguments: string[]) -> program.ExitCode {
        console.print_line(wrap<int>(1, 3).to_string());
        return program.ExitCode.Ok;
    };
);
```

```text
3
```
